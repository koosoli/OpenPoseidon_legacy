cmake_minimum_required(VERSION 3.24)
project(OpenPoseidonReShadeBridge LANGUAGES CXX)
if(NOT WIN32 OR NOT CMAKE_SIZEOF_VOID_P EQUAL 8)
    message(FATAL_ERROR "The optional ReShade bridge requires Windows x64")
endif()
if(NOT EXISTS "${RESHADE_API_DIR}/reshade.hpp")
    message(FATAL_ERROR "Set RESHADE_API_DIR to the pinned official ReShade include directory")
endif()
add_library(OpenPoseidonReShade SHARED ReShadeBridge.cpp)
target_include_directories(OpenPoseidonReShade PRIVATE "${RESHADE_API_DIR}")
target_compile_features(OpenPoseidonReShade PRIVATE cxx_std_20)
target_compile_definitions(OpenPoseidonReShade PRIVATE WIN32_LEAN_AND_MEAN NOMINMAX)
set_target_properties(OpenPoseidonReShade PROPERTIES PREFIX "" SUFFIX ".addon64")
